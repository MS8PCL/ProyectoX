const { pbAdminAuth, pbRequestAuthed } = require('./backend/src/pb.cjs');

    candStatus=`FOUND ${cand.email}`;
  }catch(e){ candStatus=`ERR ${e.status||e.statusCode||''} ${e.message}`; }

  console.log(JSON.stringify({applicationId, appStatus, importsCount:(imp.items||[]).length, candId, candStatus}, null, 2));
